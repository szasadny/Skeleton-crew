Register a runner

Once per runner, with a registration token from the repository's Settings → Actions → Runners → New self-hosted runner. See docs/install.md.

sudo install -d -o 1000 -g 1000 {{ runners_dir }}/runner-1
sudo docker run --rm -v {{ runners_dir }}/runner-1:/runner-config {{ runner_image }} \
  register --url {{ repo_url }} --token <token> \
  --labels {{ (values.get("ci.labels") or []) | join(",") }} --name skeleton-crew-runner-1
